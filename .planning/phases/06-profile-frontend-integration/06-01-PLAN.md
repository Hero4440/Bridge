<task id="3">
  <description>Add photo upload/download functions to profile.ts using Amplify Storage</description>
  
  <read_first>
    - app/src/lib/profile.ts (file we just created)
    - Look up Amplify Storage v6 uploadData and getUrl API in docs if unsure
  </read_first>
  
  <steps>
    1. Import uploadData and getUrl from 'aws-amplify/storage'
    2. Create uploadPhoto(file: File): Promise<string> function
       - Validate file.size < 5MB, throw if over
       - Validate file.type is image/jpeg, image/png, or image/webp, throw if not
       - Generate key: `profile-photos/${Date.now()}-${file.name}`
       - Call uploadData({ key, data: file }).result
       - Return result.key
    3. Create getPhotoUrl(key: string): Promise<string> function
       - Call getUrl({ key })
       - Return result.url.toString()
    4. Export both functions
  </steps>
  
  <acceptance_criteria>
    - app/src/lib/profile.ts exports uploadPhoto and getPhotoUrl functions
    - uploadPhoto validates file size (5MB max) and MIME type (jpeg/png/webp only)
    - uploadPhoto returns S3 key (string starting with "profile-photos/")
    - getPhotoUrl returns URL string
    - TypeScript compiles: `npx tsc --noEmit` passes
  </acceptance_criteria>
</task>
